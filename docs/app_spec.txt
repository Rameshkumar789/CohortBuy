<project_specification>
  <project_name>Cohort - Agentic Group Buying Platform</project_name>

  <overview>
    Build a full-stack group buying platform that aggregates consumer demand into "cohorts" and uses 
    AI agents to negotiate wholesale deals with suppliers. The platform consists of three main interfaces:
    (1) Customer side - Chrome extension for product discovery and web dashboard for pool management,
    (2) Supplier side - Portal for catalog import, inventory management, and order fulfillment,
    (3) Cohort Super Agent - 24/7 AI orchestrator that mediates between customers and suppliers.
    
    The core mechanism is "Dark Pool" commerce: buyers pledge funds (authorization hold) which are only 
    captured when the cohort target is met and a supplier accepts the deal. Prices are never published 
    publicly, allowing suppliers to offer wholesale discounts without damaging brand equity.
  </overview>

    <!-- ============================================ -->
    <cohort_super_agent>
      <orchestrator>
        - Runs 24/7 as persistent background service
        - Monitors all active pools for target completion
        - Triggers negotiation when pool reaches target quantity
        - Handles pool expiration and cleanup
        - Orchestrates payment capture after deal confirmation
        - Creates orders and notifies all parties
      </orchestrator>

      <global_catalog_agent>
        - Utility service shared by all components
        - Product search: Full-text + semantic (vector) search
        - Product matching: UPC/GTIN exact match, fuzzy title match, embedding similarity
        - Supplier lookup: Get all suppliers for a product with inventory/pricing
        - Product normalization: Use LLM to extract structured data from messy scraped input
        - Embedding generation for new products
      </global_catalog_agent>

      <negotiator_agent>
        - Spawned per-pool when negotiation triggers
        - Queries Global Catalog Agent for eligible suppliers
        - Sends RFQ (Request for Quote) to each supplier's agent
        - Collects and compares offers
        - Selects best offer based on price and availability
        - Reports result to Super Agent
        - Handles multi-round negotiation if needed
      </negotiator_agent>

      <supplier_agent>
        - One logical agent per supplier (uses supplier's config)
        - Receives RFQ from Negotiator Agent
        - Evaluates against supplier's auto-accept rules
        - AUTO mode: Immediately accept/reject based on rules
        - MANUAL mode: Create notification for human review
        - SEMI-AUTO: Auto-handle within thresholds, escalate edge cases
        - Returns response: Accept, Counter, Reject
        - Tracks open negotiations per supplier
      </supplier_agent>
    </cohort_super_agent>

      <negotiation_protocol>
        <!-- A2A-inspired structured messages for agent negotiation -->
        RFQ Message Schema:
        {
          type: 'RFQ',
          from: 'negotiator_agent',
          to: 'supplier_agent',
          payload: {
            pool_id, product_id, quantity, target_price, deadline
          }
        }

        Response Message Schema:
        {
          type: 'RFQ_RESPONSE',
          from: 'supplier_agent',
          to: 'negotiator_agent',
          payload: {
            decision: 'ACCEPT' | 'COUNTER' | 'REJECT',
            price?: number,
            reason?: string,
            valid_until?: timestamp
          }
        }

  <key_interactions>
    <pool_join_flow>
      1. User finds product (via extension or catalog browse)
      2. User clicks "Join Pool" on pool detail page
      3. If variants exist, user selects color/size
      4. Payment form appears (Stripe Elements)
      5. User enters card, clicks "Pledge $X"
      6. Backend creates PaymentIntent with manual capture
      7. Authorization hold placed on card
      8. User added to pool, count increments
      9. Real-time update pushed to all pool viewers
      10. If pool reaches target, negotiation triggers
    </pool_join_flow>

    <supplier_import_flow>
      1. Supplier uploads CSV with product data
      2. System parses CSV, extracts rows
      3. For each product:
         a. Global Catalog Agent searches for matches
         b. If match found: Show to supplier for confirmation
         c. Supplier confirms match or creates new
         d. Create supplier_catalog entry with inventory/price
      4. Import summary shown with success/error counts
    </supplier_import_flow>

    <negotiation_flow>
      1. Pool reaches target_quantity
      2. Super Agent changes pool status to NEGOTIATING
      3. Negotiator Agent spawned for pool
      4. Agent queries suppliers with inventory for product
      5. For each supplier:
         a. Send RFQ with quantity and target price
         b. Supplier Agent evaluates against rules
         c. AUTO: Immediate accept/reject
         d. MANUAL: Notify supplier, wait for response
      6. Collect all responses (with timeout)
      7. Select best offer (lowest price with available inventory)
      8. Lock deal with winning supplier
      9. Super Agent captures all payments
      10. Create orders for each pledge
      11. Notify customers and supplier
    </negotiation_flow>
  </key_interactions>
